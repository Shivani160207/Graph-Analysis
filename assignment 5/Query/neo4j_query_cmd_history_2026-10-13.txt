    AND c.courseCode IN ['CS302','CS303','CS305'])
  OR
  (d.departmentId = 'D102'
    AND c.courseCode IN ['CS301','CS304'])

MATCH (s:Student), (c:Course)

  OR (s.studentId = 'S102' AND c.courseCode IN ['CS301','CS303'])
  OR (s.studentId = 'S103' AND c.courseCode IN ['CS302','CS304'])
